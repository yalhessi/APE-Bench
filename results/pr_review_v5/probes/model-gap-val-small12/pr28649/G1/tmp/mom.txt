     9|# Moments and moment generating function
    10|
    11|## Main definitions
    12|
    13|* `ProbabilityTheory.moment X p μ`: `p`th moment of a real random variable `X` with respect to
    14|  measure `μ`, `μ[X^p]`
    15|* `ProbabilityTheory.centralMoment X p μ`:`p`th central moment of `X` with respect to measure `μ`,
    16|  `μ[(X - μ[X])^p]`
    17|* `ProbabilityTheory.mgf X μ t`: moment generating function of `X` with respect to measure `μ`,
    18|  `μ[exp(t*X)]`
    19|* `ProbabilityTheory.cgf X μ t`: cumulant generating function, logarithm of the moment generating
    20|  function
    21|
    22|## Main results
    23|
    24|* `ProbabilityTheory.IndepFun.mgf_add`: if two real random variables `X` and `Y` are independent
    25|  and their MGFs are defined at `t`, then `mgf (X + Y) μ t = mgf X μ t * mgf Y μ t`
    26|* `ProbabilityTheory.IndepFun.cgf_add`: if two real random variables `X` and `Y` are independent
    27|  and their CGFs are defined at `t`, then `cgf (X + Y) μ t = cgf X μ t + cgf Y μ t`
    28|* `ProbabilityTheory.measure_ge_le_exp_cgf` and `ProbabilityTheory.measure_le_le_exp_cgf`:
    29|  Chernoff bound on the upper (resp. lower) tail of a random variable. For `t` nonnegative such that
    30|  the cgf exists, `ℙ(ε ≤ X) ≤ exp(- t*ε + cgf X ℙ t)`. See also
    31|  `ProbabilityTheory.measure_ge_le_exp_mul_mgf` and
    32|  `ProbabilityTheory.measure_le_le_exp_mul_mgf` for versions of these results using `mgf` instead
    33|  of `cgf`.
    34|-/

    47|/-- Moment of a real random variable, `μ[X ^ p]`. -/
    48|def moment (X : Ω → ℝ) (p : ℕ) (μ : Measure Ω) : ℝ :=
    49|  μ[X ^ p]

    51|/-- Central moment of a real random variable, `μ[(X - μ[X]) ^ p]`. -/
    52|def centralMoment (X : Ω → ℝ) (p : ℕ) (μ : Measure Ω) : ℝ := by
    53|  have m := fun (x : Ω) => μ[X] -- Porting note: Lean deems `μ[(X - fun x => μ[X]) ^ p]` ambiguous
    54|  exact μ[(X - m) ^ p]

    74|    centralMoment X 1 μ = (1 - μ.real Set.univ) * μ[X] := by
    75|  simp only [centralMoment, Pi.sub_apply, pow_one]

   101|/-- Moment generating function of a real random variable `X`: `fun t => μ[exp(t*X)]`. -/
   102|def mgf (X : Ω → ℝ) (μ : Measure Ω) (t : ℝ) : ℝ :=
   103|  μ[fun ω => exp (t * X ω)]
   104|
   105|/-- Cumulant generating function of a real random variable `X`: `fun t => log μ[exp(t*X)]`. -/
   106|def cgf (X : Ω → ℝ) (μ : Measure Ω) (t : ℝ) : ℝ :=
   107|  log (mgf X μ t)

   164|theorem mgf_pos' (hμ : μ ≠ 0) (h_int_X : Integrable (fun ω => exp (t * X ω)) μ) :
   165|    0 < mgf X μ t := by
   166|  simp_rw [mgf]
   167|  have : ∫ x : Ω, exp (t * X x) ∂μ = ∫ x : Ω in Set.univ, exp (t * X x) ∂μ := by
   168|    simp only [Measure.restrict_univ]
   169|  rw [this, setIntegral_pos_iff_support_of_nonneg_ae _ _]
   170|  · have h_eq_univ : (Function.support fun x : Ω => exp (t * X x)) = Set.univ := by
   171|      ext1 x
   172|      simp only [Function.mem_support, Set.mem_univ, iff_true]
   173|      exact (exp_pos _).ne'
   174|    rw [h_eq_univ, Set.inter_univ _]
   175|    refine Ne.bot_lt ?_
   176|    simp only [hμ, ENNReal.bot_eq_zero, Ne, Measure.measure_univ_eq_zero, not_false_iff]
   177|  · filter_upwards with x
   178|    rw [Pi.zero_apply]
   179|    exact (exp_pos _).le
   180|  · rwa [integrableOn_univ]

   182|theorem mgf_pos [IsProbabilityMeasure μ] (h_int_X : Integrable (fun ω => exp (t * X ω)) μ) :
   183|    0 < mgf X μ t :=
   184|  mgf_pos' (IsProbabilityMeasure.ne_zero μ) h_int_X

   186|lemma mgf_pos_iff [hμ : NeZero μ] :
   187|    0 < mgf X μ t ↔ Integrable (fun ω ↦ exp (t * X ω)) μ := by
   188|  refine ⟨fun h ↦ ?_, fun h ↦ mgf_pos' hμ.out h⟩
   189|  contrapose! h with h

   245|/-- The moment generating function is monotone in the random variable for `t ≥ 0`. -/
   246|lemma mgf_mono_of_nonneg {Y : Ω → ℝ} (hXY : X ≤ᵐ[μ] Y) (ht : 0 ≤ t)
   247|    (htY : Integrable (fun ω ↦ exp (t * Y ω)) μ) :
   248|    mgf X μ t ≤ mgf Y μ t := by
   249|  by_cases htX : Integrable (fun ω ↦ exp (t * X ω)) μ
   250|  · refine integral_mono_ae htX htY ?_
   251|    filter_upwards [hXY] with ω hω using by gcongr
   252|  · rw [mgf_undef htX]
   253|    exact mgf_nonneg
   254|
   255|/-- The moment generating function is antitone in the random variable for `t ≤ 0`. -/
   256|lemma mgf_anti_of_nonpos {Y : Ω → ℝ} (hXY : X ≤ᵐ[μ] Y) (ht : t ≤ 0)
   257|    (htX : Integrable (fun ω ↦ exp (t * X ω)) μ) :
   258|    mgf Y μ t ≤ mgf X μ t := by
   259|  by_cases htY : Integrable (fun ω ↦ exp (t * Y ω)) μ
   260|  · refine integral_mono_ae htY htX ?_
   261|    filter_upwards [hXY] with ω hω using exp_monotone <| mul_le_mul_of_nonpos_left hω ht
   262|  · rw [mgf_undef htY]
   263|    exact mgf_nonneg

   267|/-- This is a trivial application of `IndepFun.comp` but it will come up frequently. -/
   268|theorem IndepFun.exp_mul {X Y : Ω → ℝ} (h_indep : IndepFun X Y μ) (s t : ℝ) :

   271|  change IndepFun ((fun x => exp (s * x)) ∘ X) ((fun x => exp (t * x)) ∘ Y) μ
   272|  exact IndepFun.comp h_indep (h_meas s) (h_meas t)

   291|    (h_int_X : Integrable (fun ω => exp (t * X ω)) μ)

   392|/-- **Chernoff bound** on the upper tail of a real random variable. -/
   393|theorem measure_ge_le_exp_mul_mgf [IsFiniteMeasure μ] (ε : ℝ) (ht : 0 ≤ t)
   394|    (h_int : Integrable (fun ω => exp (t * X ω)) μ) :
   395|    μ.real {ω | ε ≤ X ω} ≤ exp (-t * ε) * mgf X μ t := by
   396|  rcases ht.eq_or_lt with ht_zero_eq | ht_pos
   397|  · rw [ht_zero_eq.symm]
   398|    simp only [neg_zero, zero_mul, exp_zero, mgf_zero', one_mul]
   399|    gcongr
   400|    exacts [measure_ne_top _ _, Set.subset_univ _]
   401|  calc
   402|    μ.real {ω | ε ≤ X ω} = μ.real {ω | exp (t * ε) ≤ exp (t * X ω)} := by
   403|      congr 1 with ω
   404|      simp only [Set.mem_setOf_eq, exp_le_exp]
   405|      exact ⟨fun h => mul_le_mul_of_nonneg_left h ht_pos.le,
   406|        fun h => le_of_mul_le_mul_left h ht_pos⟩
   407|    _ ≤ (exp (t * ε))⁻¹ * μ[fun ω => exp (t * X ω)] := by
   408|      have : exp (t * ε) * μ.real {ω | exp (t * ε) ≤ exp (t * X ω)} ≤
   409|          μ[fun ω => exp (t * X ω)] :=
   410|        mul_meas_ge_le_integral_of_nonneg (ae_of_all _ fun x => (exp_pos _).le) h_int _
   411|      rwa [mul_comm (exp (t * ε))⁻¹, ← div_eq_mul_inv, le_div_iff₀' (exp_pos _)]
   412|    _ = exp (-t * ε) * mgf X μ t := by rw [neg_mul, exp_neg]; rfl

   415|theorem measure_le_le_exp_mul_mgf [IsFiniteMeasure μ] (ε : ℝ) (ht : t ≤ 0)
   416|    (h_int : Integrable (fun ω => exp (t * X ω)) μ) :
